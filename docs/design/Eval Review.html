// Native behaviour that CSS cannot do. Loaded from web/layout.templ as an ES module:
//   <script type="module" src="/static/native.js"></script>
// Plain JavaScript: no framework, no build step, and nothing a strict Content-Security-Policy
// (script-src 'self') blocks. Everything else that makes the app feel native is in static/app.css.

// Press feedback on iPhone and iPad. WebKit on iOS shows :active styles only when a touch listener
// sits on the element or on <body>. Without this, the pressed look of the tabs and the Search button
// would never show, and with the grey tap flash turned off in app.css a tap would give no feedback
// at all. Empty passive listeners on <body> and document switch :active on without delaying
// scrolling. (A module runs after the page is parsed, so <body> exists.) iOS still drops :active
// for a touch that lands during momentum scrolling; that is the platform, not the app.
const noop = () => {};
document.body.addEventListener("touchstart", noop, { passive: true });
document.addEventListener("touchstart", noop, { passive: true });
